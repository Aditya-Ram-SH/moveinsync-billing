"""
CORE BILLING ENGINE
This service unifies all 4 contract models into a single calculation flow.
"""
from sqlalchemy.orm import Session
from sqlalchemy import func
from datetime import datetime, time
from app import models
from decimal import Decimal

def calculate_trip_cost(trip: models.Trip, contract: models.Contract):
    """
    Apply Contract Rules to a single Trip.
    Returns: {final_cost, breakdown}
    """
    config = contract.config_json
    model_type = contract.model_type # PACKAGE, TRIP, HYBRID
    
    # 1. PARSE TIMING (Night Shift Logic)
    # Trip Schema: pickup_time, dropoff_time
    is_night = False
    pickup_hour = trip.pickup_time.hour
    if pickup_hour >= 22 or pickup_hour < 6:
        is_night = True

    # 2. PARSE DELAY (Employee Incentive Logic)
    # Trip Schema: booking_time (scheduled), pickup_time (actual)
    delay_minutes = (trip.pickup_time - trip.booking_time).total_seconds() / 60
    employee_incentive = 0
    
    # Check Employee Incentive Rules
    emp_rules = config.get("employee_incentives", {})
    if delay_minutes > emp_rules.get("delay_threshold_min", 999):
        employee_incentive = Decimal(str(emp_rules.get("delay_compensation_amount", 0)))

    # 3. CALCULATE VENDOR COST
    vendor_cost = 0
    
    # --- LOGIC: PACKAGE MODEL ---
    if model_type == "PACKAGE":
        # In Package model, individual trips are usually $0 (covered by fixed fee)
        # UNLESS there are specific per-trip surcharges defined
        vendor_cost = 0 
        
        # Check Vendor Performance Pay (Overage/Hardship)
        vendor_rules = config.get("vendor_payouts", {})
        if is_night:
             vendor_cost += Decimal(str(vendor_rules.get("night_shift_bonus", 0)))

    # --- LOGIC: TRIP MODEL ---
    elif model_type == "TRIP":
        rates = config.get("rates", {})
        base = Decimal(str(rates.get("base_fare", 0)))
        per_km = Decimal(str(rates.get("per_km_rate", 0)))
        night_mult = Decimal(str(rates.get("night_multiplier", 1.0)))
        
        # Math
        ride_cost = base + (trip.distance_km * per_km)
        
        if is_night:
            ride_cost *= night_mult
            
        vendor_cost = ride_cost

    # --- LOGIC: HYBRID (BASE TIER) ---
    elif model_type == "HYBRID":
        rates = config.get("rates", {})
        thresholds = config.get("thresholds", {})
        
        fixed_base = Decimal(str(rates.get("fixed_base_pay", 0)))
        included_km = Decimal(str(thresholds.get("included_km_per_trip", 0)))
        per_extra_km = Decimal(str(rates.get("per_extra_km_rate", 0)))
        
        # Base is guaranteed
        vendor_cost = fixed_base
        
        # Add Overage
        if trip.distance_km > included_km:
            extra_km = trip.distance_km - included_km
            vendor_cost += (extra_km * per_extra_km)

    return {
        "vendor_cost": vendor_cost,
        "employee_incentive": employee_incentive,
        "is_night": is_night,
        "delay_min": delay_minutes
    }

def run_billing_run(db: Session, client_id: int, vendor_id: int, month: datetime.date):
    """
    The 'Generate Now' Button Logic.
    """
    # 1. Fetch Contract
    contract = db.query(models.Contract).filter(
        models.Contract.client_id == client_id,
        models.Contract.vendor_id == vendor_id,
        models.Contract.is_active == True
    ).first()
    
    if not contract:
        raise Exception("No Active Contract found")

    # 2. Fetch Trips for Month
    # (Assuming we have a helper to get start/end of month)
    trips = db.query(models.Trip).filter(
        models.Trip.client_id == client_id,
        models.Trip.vendor_id == vendor_id,
        # Filter by date range logic here...
        models.Trip.status == "INGESTED"
    ).all()
    
    total_bill = 0
    
    # 3. Loop and Compute
    for trip in trips:
        result = calculate_trip_cost(trip, contract)
        
        # 4. Save Charge
        charge = models.TripCharge(
            trip_id=trip.trip_id,
            final_cost=result["vendor_cost"], # Simplified for demo
            formula_snapshot=contract.config_json
            # Add employee incentive column if you added it to schema
        )
        db.add(charge)
        
        # 5. Update Trip
        trip.status = "PROCESSED"
        
        total_bill += result["vendor_cost"]

    # 6. Handle PACKAGE Fixed Fee (Once per month)
    if contract.model_type == "PACKAGE":
        fixed_fee = Decimal(str(contract.config_json.get("monthly_fixed_pay", 0)))
        total_bill += fixed_fee
        # You might insert a special "Fixed Fee" charge row here

    return {"status": "SUCCESS", "total": total_bill}




1. The Logic Flow
Trigger: Admin selects "November 2025" and clicks "Generate".

Validation: System compares Today vs. Month End Date.

If Today < Month End: System pauses and shows a Modal: "Warning: November is not over. This will generate a Provisional Bill for tracking purposes only."

Action: Admin clicks "Proceed".

Execution: System calculates cost for trips up to this exact moment.

Result: The Dashboard shows the bill with a Yellow Badge: PROVISIONAL / DRAFT.

2. The "Overwrite" Rule (Crucial)
Since your database has a UNIQUE constraint on (client, vendor, month), you need a smart rule for Provisional bills:

Final Bills are Immutable: Once a bill is marked SUCCESS (Final), it cannot be touched. The month is closed.

Provisional Bills are Overwritable: If I generate a Provisional Bill on Nov 24, and then run it again on Nov 25, the system should automatically delete/update the Nov 24 entry.

Why: Because the first one was just an estimate. We don't want "Duplicate Key" errors blocking the team from checking updated costs.
